cmake_minimum_required(VERSION 3.16)
project(waveform_rtl433 C)

if(NOT EXISTS "${RTL433_SOURCE}/include/pulse_detect.h")
  message(FATAL_ERROR "Set RTL433_SOURCE to the reviewed rtl_433 source tree")
endif()
set(ENABLE_RTLSDR OFF CACHE STRING "" FORCE)
set(ENABLE_SOAPYSDR OFF CACHE STRING "" FORCE)
set(ENABLE_OPENSSL OFF CACHE STRING "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory("${RTL433_SOURCE}" rtl433 EXCLUDE_FROM_ALL)
add_executable(waveform_rtl433 main.c)
target_compile_features(waveform_rtl433 PRIVATE c_std_99)
target_compile_options(waveform_rtl433 PRIVATE -Wall -Wextra -Werror)
target_include_directories(waveform_rtl433 PRIVATE "${RTL433_SOURCE}/include")
target_link_libraries(waveform_rtl433 PRIVATE r_433 m)
